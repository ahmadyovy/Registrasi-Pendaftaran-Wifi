    <script>
        // ImgBB Free API Key
        const IMGBB_API_KEY = "c18c5e9334c4f6990499e9c5a1dfd9a0";

        // State data
        let selectedPackage = {
            name: '100Mbps',
            price: '220.890'
        };

        let uploadedUrls = {
            ktp: '',
            house: ''
        };

        // Initialize default UI selection
        window.onload = function() {
            const firstCard = document.querySelector('.package-card');
            if (firstCard) firstCard.classList.add('selected');
        };

        // Select Package Logic
        function selectPackage(name, price, element) {
            selectedPackage = { name, price };
            
            document.querySelectorAll('.package-card').forEach(card => {
                card.classList.remove('selected');
                card.querySelector('input[type="radio"]').checked = false;
            });

            element.classList.add('selected');
            element.querySelector('input[type="radio"]').checked = true;
        }

        // Upload Image to ImgBB Cloud API with robust fallback
        async function handleFileUpload(input, previewContainerId, fileNameId, urlTypeKey) {
            const previewContainer = document.getElementById(previewContainerId);
            const defaultUI = document.getElementById(previewContainerId.replace('Preview', 'DefaultUI'));
            const loadingUI = document.getElementById(previewContainerId.replace('Preview', 'Loading'));
            const imgElement = previewContainer.querySelector('img');

            if (input.files && input.files[0]) {
                const file = input.files[0];

                // Show local preview immediately
                const reader = new FileReader();
                reader.onload = function(e) {
                    imgElement.src = e.target.result;
                }
                reader.readAsDataURL(file);

                // Show Loading UI
                defaultUI.classList.add('hidden');
                previewContainer.classList.add('hidden');
                loadingUI.classList.remove('hidden');

                // Upload via FormData to ImgBB
                const formData = new FormData();
                formData.append('image', file);

                try {
                    const response = await fetch(`https://api.imgbb.com/1/upload?key=${IMGBB_API_KEY}`, {
                        method: 'POST',
                        body: formData
                    });

                    const result = await response.json();

                    if (result.success) {
                        if (urlTypeKey === 'ktpUrl') uploadedUrls.ktp = result.data.url;
                        if (urlTypeKey === 'houseUrl') uploadedUrls.house = result.data.url;

                        loadingUI.classList.add('hidden');
                        previewContainer.classList.remove('hidden');
                    } else {
                        // Fallback jika API menolak, tetap tandai foto sudah dipilih agar tidak gagal total
                        if (urlTypeKey === 'ktpUrl') uploadedUrls.ktp = '✅ (Foto KTP Berhasil Dipilih di HP)';
                        if (urlTypeKey === 'houseUrl') uploadedUrls.house = '✅ (Foto Rumah Berhasil Dipilih di HP)';
                        loadingUI.classList.add('hidden');
                        previewContainer.classList.remove('hidden');
                    }
                } catch (error) {
                    console.error("Error uploading image:", error);
                    // Fallback offline / network block / CORS
                    if (urlTypeKey === 'ktpUrl') uploadedUrls.ktp = '✅ (Foto KTP Berhasil Dipilih di HP)';
                    if (urlTypeKey === 'houseUrl') uploadedUrls.house = '✅ (Foto Rumah Berhasil Dipilih di HP)';
                    loadingUI.classList.add('hidden');
                    previewContainer.classList.remove('hidden');
                }
            }
        }

        // Form Submission Handler
        function submitForm(e) {
            e.preventDefault();

            const fullName = document.getElementById('fullName').value.trim();
            const phoneNumber = document.getElementById('phoneNumber').value.trim();
            const email = document.getElementById('email').value.trim();
            const address = document.getElementById('address').value.trim();
            const referralCode = document.getElementById('referralCode').value.trim();

            const ktpInputFile = document.getElementById('ktpInput').files[0];
            const houseInputFile = document.getElementById('houseInput').files[0];

            const ktpLink = uploadedUrls.ktp ? uploadedUrls.ktp : (ktpInputFile ? '✅ (Foto KTP Telah Dipilih)' : '❌ (Belum diupload)');
            const houseLink = uploadedUrls.house ? uploadedUrls.house : (houseInputFile ? '✅ (Foto Rumah Telp Dipilih)' : '❌ (Belum diupload)');

            // Formatted Message Text for WhatsApp
            const message = 
`Halo Kak Yovy, saya ingin mendaftar Indosat HiFi. Berikut data pendaftaran saya:

📋 *DETAIL FORMULIR HIFI*
• *Paket Dipilih:* Speed ${selectedPackage.name} (Rp ${selectedPackage.price}/bulan)
• *Nama Lengkap:* ${fullName}
• *No. HP / WA:* ${phoneNumber}
• *Email:* ${email}
• *Alamat Pemasangan:* ${address}
• *Kode Referral:* ${referralCode ? referralCode.toUpperCase() : '-'}

📸 *LINK FOTO LAMPIRAN:*
• *Foto KTP:* ${ktpLink}
• *Foto Tampak Depan Rumah:* ${houseLink}

*(Seluruh harga paket sudah termasuk PPN 11%)*

Mohon bantuannya untuk diproses cek coverage dan pendaftarannya ya Kak Yovy, terima kasih!`;

            const encodedText = encodeURIComponent(message);
            const targetPhone = "6282258659397"; // Nomor WA Kak Yovy
            const whatsappUrl = `https://api.whatsapp.com/send?phone=${targetPhone}&text=${encodedText}`;

            // Directly open WhatsApp
            window.location.href = whatsappUrl;
        }
